const transportChars=['车','地','共','汽','校','出','电','租','铁','车','公','火','车','车','车'];
let activeTransport=0;
let transportBuilds=[];
let transportBankState=[];
function renderTransport(){
  const h=document.getElementById('transport');
  const bank=document.getElementById('transportBank');
  h.innerHTML=''; bank.innerHTML='';
  transportBuilds=trans.map(()=>[]);
  transportBankState=transportChars.map((ch,id)=>({ch,id,used:false}));
  trans.forEach((x,i)=>{
    const d=document.createElement('div');
    d.className='transport-card'+(i===0?' active':'');
    d.dataset.i=i;
    d.innerHTML=`<img src="${x.img}" alt="Транспорт ${i+1}"><span class="badge">${i+1}</span><div class="build-word" data-i="${i}" aria-label="Поле ответа ${i+1}"></div><div class="feedback"></div>`;
    d.onclick=(ev)=>{if(!ev.target.classList.contains('built-char'))setActiveTransport(i)};
    const zone=d.querySelector('.build-word');
    zone.ondragover=e=>e.preventDefault();
    zone.ondrop=e=>{e.preventDefault();const id=Number(e.dataTransfer.getData('text/plain'));if(Number.isInteger(id))addTransportChar(i,id)};
    h.appendChild(d);
  });
  transportBankState.forEach(item=>{
    const b=document.createElement('button');
    b.type='button'; b.className='transport-char'; b.textContent=item.ch; b.dataset.id=item.id; b.draggable=true;
    b.onclick=()=>addTransportChar(activeTransport,item.id);
    b.ondragstart=e=>e.dataTransfer.setData('text/plain',String(item.id));
    bank.appendChild(b);
  });
  refreshTransportUI();
}
function setActiveTransport(i){activeTransport=i;document.querySelectorAll('.transport-card').forEach((c,j)=>c.classList.toggle('active',j===i))}
function addTransportChar(cardIndex,charId){
  const item=transportBankState[charId]; if(!item||item.used)return;
  item.used=true; transportBuilds[cardIndex].push(charId); activeTransport=cardIndex; refreshTransportUI();
}
function removeTransportChar(cardIndex,pos){
  const id=transportBuilds[cardIndex][pos]; if(id===undefined)return;
  transportBuilds[cardIndex].splice(pos,1); if(transportBankState[id])transportBankState[id].used=false; activeTransport=cardIndex; refreshTransportUI();
}
function refreshTransportUI(){
  document.querySelectorAll('.transport-card').forEach((card,i)=>{
    card.classList.toggle('active',i===activeTransport);
    const zone=card.querySelector('.build-word'); zone.innerHTML='';
    transportBuilds[i].forEach((id,pos)=>{const b=document.createElement('button');b.type='button';b.className='built-char';b.textContent=transportBankState[id].ch;b.title='Нажмите, чтобы вернуть иероглиф';b.onclick=e=>{e.stopPropagation();removeTransportChar(i,pos)};zone.appendChild(b)});
  });
  document.querySelectorAll('.transport-char').forEach(b=>{const id=Number(b.dataset.id);b.classList.toggle('used',!!transportBankState[id]?.used)});
}
function checkTransport(){
  let c=0;
  document.querySelectorAll('.transport-card').forEach((d,i)=>{
    const word=transportBuilds[i].map(id=>transportBankState[id].ch).join('');
    const ok=word===trans[i].ans;
    const f=d.querySelector('.feedback'); f.className='feedback '+(ok?'ok':'bad'); f.textContent=ok?'✓ Верно':'✗ Проверьте порядок иероглифов';
    if(ok)c++;
  });
  const all=c===trans.length; markComplete(4,all); showResult('res4',all,`Правильно собрано: ${c} / ${trans.length}`);
}
function resetTransport(){renderTransport();markComplete(4,false);showResult('res4',false,'Все иероглифы возвращены в панель справа.')}
renderTransport();
